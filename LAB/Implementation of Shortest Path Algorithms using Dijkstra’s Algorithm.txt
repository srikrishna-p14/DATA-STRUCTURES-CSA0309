#include <stdio.h>
#define INF 9999

int main() {
    int cost[20][20], dist[20], visited[20];
    int n, i, j, start, count, u, min;

    printf("Enter number of vertices: ");
    scanf("%d", &n);

    printf("Enter cost matrix (0 for no edge):\n");
    for (i = 0; i < n; i++)
        for (j = 0; j < n; j++) {
            scanf("%d", &cost[i][j]);
            if (cost[i][j] == 0 && i != j)
                cost[i][j] = INF;
        }

    printf("Enter source vertex: ");
    scanf("%d", &start);

    for (i = 0; i < n; i++) {
        dist[i] = cost[start][i];
        visited[i] = 0;
    }
    dist[start] = 0;
    visited[start] = 1;
    for (count = 1; count < n; count++) {
        min = INF;
        u = -1;
        for (i = 0; i < n; i++) {
            if (!visited[i] && dist[i] < min) {
                min = dist[i];
                u = i;
            }
        }
        if (u == -1)
            break;
        visited[u] = 1;
        for (i = 0; i < n; i++) {
            if (!visited[i] &&
                dist[u] + cost[u][i] < dist[i])
                dist[i] = dist[u] + cost[u][i];
        }
    }
    printf("Shortest distances:\n");
    for (i = 0; i < n; i++)
        printf("%d -> %d = %d\n", start, i, dist[i]);
    return 0;
}
