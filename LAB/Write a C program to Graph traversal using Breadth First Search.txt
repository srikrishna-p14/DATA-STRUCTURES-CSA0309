#include <stdio.h>
int main() {
    int a[20][20], visited[20] = {0};
    int q[20], front = 0, rear = 0;
    int n, i, j, start;
    printf("Enter number of vertices: ");
    scanf("%d", &n);
    printf("Enter adjacency matrix:\n");
    for (i = 0; i < n; i++)
        for (j = 0; j < n; j++)
            scanf("%d", &a[i][j]);
    printf("Enter starting vertex: ");
    scanf("%d", &start);
    q[rear++] = start;
    visited[start] = 1;
    printf("BFS: ");
    while (front < rear) {
        int v = q[front++];
        printf("%d ", v);
        for (i = 0; i < n; i++) {
            if (a[v][i] && !visited[i]) {
                q[rear++] = i;
                visited[i] = 1;
            }
        }
    }
    return 0;
}
