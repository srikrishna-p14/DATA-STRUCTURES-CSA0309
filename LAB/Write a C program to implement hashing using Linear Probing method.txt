#include <stdio.h>
#define SIZE 10
int hash[SIZE];
void insert(int key) {
    int index = key % SIZE;
    int start = index;

    while (hash[index] != -1) {
        index = (index + 1) % SIZE;

        if (index == start) {
            printf("Hash table full\n");
            return;
        }
    }
    hash[index] = key;
}
void display() {
    int i;

    for (i = 0; i < SIZE; i++) {
        if (hash[i] == -1)
            printf("%d : Empty\n", i);
        else
            printf("%d : %d\n", i, hash[i]);
    }
}
int main() {
    int i, n, x;

    for (i = 0; i < SIZE; i++)
        hash[i] = -1;
    printf("Enter number of elements: ");
    scanf("%d", &n);

    for (i = 0; i < n; i++) {
        scanf("%d", &x);
        insert(x);
    }
    display();

    return 0;
}
